Automated Input Injection & Output Truth Table Verification Script

Automatically feeds truth table inputs (A3..A0, B3..B0) into the circuit, captures the live gate-level outputs (CARRY, S3..S0), verifies them against the expected binary arithmetic truth table, and steps at your configured time_delay (configured to 200s by default, with instant step & fast sweep options).

Automated Truth Table Verification Script (`time_delay = 200s`)

Runs directly inside this simulator AND works in the CircuitVerse browser console (`F12 → Console`).

// Automated Truth Table Verification Script for CO_MINI_PROJECT_GROUP_4 (4-Bit Binary Adder)
// Automatically feeds 9-bit inputs (A3..A0, B3..B0), reads 5-bit outputs (CARRY, S3..S0), and verifies truth table.
const TIME_DELAY_SECONDS = 200; // Time delay = 200s (200,000 ms) between each input vector
const TIME_DELAY_MS = TIME_DELAY_SECONDS * 1000;

async function autoVerifyTruthTable(delayMs = TIME_DELAY_MS) {
  console.log(`Starting Truth Table Verification with delay = ${delayMs / 1000}s (${delayMs}ms)...`);
  let passed = 0, failed = 0, step = 0;

  for (let M = 0; M <= 1; M++) {
    for (let A = 0; A < 16; A++) {
      for (let B = 0; B < 16; B++) {
        step++;
        // 1. Automatically apply input bits to circuit pins (A3..A0, B3..B0, M)
        window.setCircuitInputs(A, B, M);

        // 2. Read actual gate-level outputs from the circuit simulator
        const out = window.readCircuitOutputs(); // { CARRY, S3, S2, S1, S0, valS, val5Bit }

        // 3. Compute expected truth table output using B = B ^ (M ? 0xF : 0x0) and C0 = 0
        const bComp = M === 1 ? (B ^ 0xF) : B;
        const expected5Bit = A + bComp + M;
        const expCarry = (expected5Bit >> 4) & 1;
        const expS = expected5Bit & 0xF;

        const isPass = (out.CARRY === expCarry) && (out.valS === expS);
        if (isPass) passed++; else failed++;

        console.log(
          `[Step ${step}/512] M=${M} | A=${A.toString(2).padStart(4,'0')} (${A}) | B=${B.toString(2).padStart(4,'0')} (${B}) ` +
          `=> CARRY=${out.CARRY} S=${out.valS.toString(2).padStart(4,'0')} | Expected=${expCarry}_${expS.toString(2).padStart(4,'0')} | ${isPass ? 'PASS ✅' : 'FAIL ❌'}`
        );

        // 4. Wait configured time delay (200s = 200,000ms) before feeding next input
        await new Promise(resolve => setTimeout(resolve, delayMs));
      }
    }
  }
  console.log(`Verification Complete! Passed: ${passed}, Failed: ${failed}`);
}
